<script type="module">
import { createClient } from 'https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2/+esm';

/* -------------------------------------------------------
   1. ADD YOUR SUPABASE PROJECT DETAILS HERE.
   Use your Project URL and publishable key.
   NEVER put a service_role or secret key in this file.
-------------------------------------------------------- */
const SUPABASE_URL = 'PASTE_YOUR_PROJECT_URL_HERE';
const SUPABASE_KEY = 'PASTE_YOUR_PUBLISHABLE_KEY_HERE';

const rooms = {
  global:   { name: 'Global Lounge', icon: '🌎', subtitle: 'A place to meet people worldwide' },
  english:  { name: 'English Chat', icon: '💬', subtitle: 'Talk and practise English' },
  asia:     { name: 'Asia Pacific', icon: '🌏', subtitle: 'Connect across Asia Pacific' },
  europe:   { name: 'Europe', icon: '🏰', subtitle: 'Meet people from Europe' },
  americas: { name: 'Americas', icon: '🌎', subtitle: 'North and South America' },
  gaming:   { name: 'Gaming Zone', icon: '🎮', subtitle: 'Games, players and teammates' },
  music:    { name: 'Music & Culture', icon: '🎵', subtitle: 'Talk about music and culture' },
  language: { name: 'Language Exchange', icon: '🗣️', subtitle: 'Learn languages together' }
};

const $ = id => document.getElementById(id);
let supabase = null;
let currentRoom = 'global';
let channel = null;
let roomRequest = 0;
let seenIds = new Set();
let connected = false;
let profile = {
  username: localStorage.getItem('vybe_username') || '',
  country: localStorage.getItem('vybe_country') || '🇦🇺'
};

$('username').value = profile.username;
$('country').value = profile.country;

function toast(message) {
  const el = $('toast');
  el.textContent = message;
  el.classList.add('show');
  clearTimeout(toast.timer);
  toast.timer = setTimeout(() => el.classList.remove('show'), 3500);
}

function setStatus(text, state = '') {
  $('statusText').textContent = text;
  $('statusDot').className = 'dot ' + state;
}

function escapeText(value) {
  return String(value ?? '').replace(/[&<>"']/g, ch => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  })[ch]);
}

function timeLabel(value) {
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return '';
  return date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
}

function addMessage(message) {
  if (!message || seenIds.has(message.id)) return;
  seenIds.add(message.id);

  const note = $('loadingNote');
  if (note) note.remove();

  const row = document.createElement('article');
  row.className = 'message';
  const username = escapeText(message.username || 'Guest');
  const country = escapeText(message.country || '🌍');
  const body = escapeText(message.message || '');
  const time = escapeText(timeLabel(message.created_at));

  row.innerHTML = `
    <div class="avatar">${escapeText((message.username || 'G').trim().charAt(0).toUpperCase() || 'G')}</div>
    <div class="msg-content">
      <div class="msg-head">
        <span class="msg-name">${country} ${username}</span>
        <span class="msg-time">${time}</span>
      </div>
      <div class="msg-text">${body}</div>
    </div>`;

  $('messages').appendChild(row);
  $('messages').scrollTop = $('messages').scrollHeight;
}

function showNote(message) {
  $('messages').innerHTML = `
    <div class="welcome">
      <div class="eyebrow">WELCOME TO VYBE GLOBAL</div>
      <h3>Different places. One conversation.</h3>
      <p>Meet people, talk about your interests and discover what is happening around the world.</p>
    </div>
    <div class="system-note" id="loadingNote">${escapeText(message)}</div>`;
  seenIds.clear();
}

function saveProfile() {
  const username = $('username').value.trim();
  if (username.length < 2) {
    toast('Choose a username with at least 2 characters.');
    return false;
  }
  profile.username = username.slice(0, 25);
  profile.country = $('country').value || '🌍';
  localStorage.setItem('vybe_username', profile.username);
  localStorage.setItem('vybe_country', profile.country);
  toast('Profile saved on this device.');
  return true;
}

$('saveProfile').addEventListener('click', saveProfile);

function configured() {
  return SUPABASE_URL.startsWith('https://')
    && !SUPABASE_URL.includes('PASTE_')
    && SUPABASE_KEY.length > 20
    && !SUPABASE_KEY.includes('PASTE_');
}

async function openRoom(roomId) {
  if (!rooms[roomId]) return;
  currentRoom = roomId;
  const request = ++roomRequest;
  const room = rooms[roomId];

  document.querySelectorAll('.room').forEach(button => {
    button.classList.toggle('active', button.dataset.room === roomId);
  });

  $('topIcon').textContent = room.icon;
  $('roomTitle').textContent = room.name;
  $('roomSubtitle').textContent = room.subtitle;
  $('sidebar').classList.remove('open');
  showNote('Loading messages...');

  if (!supabase) {
    setStatus('Setup needed', 'error');
    showNote('Add your Supabase URL and publishable key in the code, then refresh.');
    return;
  }

  if (channel) {
    const oldChannel = channel;
    channel = null;
    await supabase.removeChannel(oldChannel);
  }
  if (request !== roomRequest) return;

  setStatus('Connecting');
  const { data, error } = await supabase
    .from('chat_messages')
    .select('id, room, username, country, message, created_at')
    .eq('room', roomId)
    .order('created_at', { ascending: true })
    .limit(100);

  if (request !== roomRequest) return;
  if (error) {
    console.error(error);
    setStatus('Connection error', 'error');
    showNote('Could not load messages. Check your Supabase table, policies and internet connection.');
    return;
  }

  showNote(data.length ? '' : 'No messages yet. Be the first to say hello!');
  for (const item of data) addMessage(item);

  // Listen for new messages from other devices in this room.
  channel = supabase
    .channel('vybe-room-' + roomId + '-' + request)
    .on('postgres_changes', {
      event: 'INSERT',
      schema: 'public',
      table: 'chat_messages',
      filter: 'room=eq.' + roomId
    }, payload => {
      if (currentRoom === roomId) addMessage(payload.new);
    })
    .subscribe(status => {
      if (request !== roomRequest) return;
      if (status === 'SUBSCRIBED') {
        connected = true;
        setStatus('Live chat', 'connected');
      } else if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT') {
        connected = false;
        setStatus('Realtime issue', 'error');
        console.error('Supabase Realtime status:', status);
      }
    });
}

document.querySelectorAll('.room').forEach(button => {
  button.addEventListener('click', () => openRoom(button.dataset.room));
});

$('menuBtn').addEventListener('click', () => {
  $('sidebar').classList.toggle('open');
});

$('messageInput').addEventListener('input', () => {
  $('charCount').textContent = $('messageInput').value.length + ' / 1000';
});

$('chatForm').addEventListener('submit', async event => {
  event.preventDefault();
  if (!supabase) {
    toast('First add your Supabase URL and publishable key.');
    return;
  }
  if (!profile.username || profile.username.trim().length < 2) {
    toast('Set your username in Your Profile first.');
    $('username').focus();
    return;
  }

  const message = $('messageInput').value.trim();
  if (!message) return;

  const sendButton = $('sendBtn');
  sendButton.disabled = true;
  sendButton.textContent = 'SENDING...';

  const { error } = await supabase.from('chat_messages').insert({
    room: currentRoom,
    username: profile.username,
    country: profile.country,
    message: message.slice(0, 1000)
  });

  sendButton.disabled = false;
  sendButton.textContent = 'SEND ↗';

  if (error) {
    console.error(error);
    toast('Message failed. Check your Supabase setup and table policies.');
    return;
  }

  $('messageInput').value = '';
  $('charCount').textContent = '0 / 1000';
  $('messageInput').focus();
});

if (configured()) {
  try {
    supabase = createClient(SUPABASE_URL, SUPABASE_KEY);
    setStatus('Connecting');
    openRoom(currentRoom);
  } catch (error) {
    console.error(error);
    setStatus('Setup error', 'error');
    showNote('Could not initialise Supabase. Check your project URL and publishable key.');
  }
} else {
  setStatus('Setup needed', 'error');
  showNote('Step 1: Create your Supabase project. Step 2: Run the SQL setup. Step 3: Paste your URL and publishable key into this file.');
}
</script>
